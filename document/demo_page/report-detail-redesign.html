คำสั่ง SQL

มองเห็นเฉพาะผู้ดูแลระบบ
ดึงยอดขายรายเดือน MAIN v1.4
1SELECT region, product_group,
2 SUM(sale_amount) AS total_sales
3FROM sales_transactions
4WHERE sale_month = :month
5GROUP BY region, product_group
6ORDER BY total_sales DESC;
พารามิเตอร์
Paramชนิดค่าเริ่มต้นจำเป็นเงื่อนไขที่ใช้
:month date เดือนปัจจุบัน ต้องระบุ sale_month = :month
คำนวณส่วนต่างเป้าหมาย v1.0
1SELECT s.region,
2 s.total_sales - t.target_amount AS variance
3FROM monthly_sales s
4JOIN sales_targets t ON t.region = s.region;

ไม่มีพารามิเตอร์สำหรับคำสั่งนี้

ตรวจสอบรายละเอียดรายการขายทั้งหมด v2.1
1SELECT t.transaction_id, t.sale_date, t.region, t.product_group, t.product_name, t.quantity, t.unit_price, t.sale_amount, c.customer_name, c.customer_tier, e.employee_name, e.branch_code
2FROM sales_transactions t
3JOIN customers c ON c.customer_id = t.customer_id
4JOIN employees e ON e.employee_id = t.employee_id
5WHERE t.sale_date BETWEEN :date_from AND :date_to
6  AND t.sale_amount >= :min_amount
7  AND (:region IS NULL OR t.region = :region)
8ORDER BY t.sale_date DESC, t.transaction_id DESC
9LIMIT 500;
พารามิเตอร์
Paramชนิดค่าเริ่มต้นจำเป็นเงื่อนไขที่ใช้
:date_from date – ต้องระบุ t.sale_date >= :date_from
:date_to date – ต้องระบุ t.sale_date <= :date_to
:min_amount number 0 ไม่บังคับ t.sale_amount >= :min_amount
:region string – ไม่บังคับ t.region = :region (กรองเฉพาะเมื่อระบุ)
